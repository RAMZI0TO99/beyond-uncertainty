
def _wire(value):
    return (json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False)+'\n').encode('utf8')


def _time(second):
    return '2026-10-25T00:00:'+str(second).zfill(2)+'.000000+00:00'


def _ref(label):
    return ('$REF',label)


def _pin(path,raw):
    return dict(path=path,bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())


def _fingerprint(value):
    def describe(v):
        if type(v) is bytes:return ['bytes',len(v),hashlib.sha256(v).hexdigest()]
        if type(v) in (str,int,bool) or v is None:return [type(v).__name__,v]
        if type(v) is dict:return ['dict',[[describe(k),describe(x)] for k,x in sorted(v.items())]]
        if type(v) in (list,tuple):return [type(v).__name__,[describe(x) for x in v]]
        if isinstance(v,dict):return ['dict_subclass',describe(dict(v))]
        if isinstance(v,list):return ['list_subclass',describe(list(v))]
        if isinstance(v,str):return ['str_subclass',str(v)]
        if isinstance(v,int):return ['int_subclass',int(v)]
        if isinstance(v,bytes):return ['bytes_subclass',len(v),hashlib.sha256(v).hexdigest()]
        return [type(v).__name__,repr(v)]
    return hashlib.sha256(_wire(describe(value))).hexdigest()


def _mutable_ids(value,seen=None):
    seen=set() if seen is None else seen
    if type(value) in (dict,list,set):
        if id(value) in seen:return seen
        seen.add(id(value))
        children=value.values() if type(value) is dict else value
        for child in children:_mutable_ids(child,seen)
    elif type(value) is tuple:
        for child in value:_mutable_ids(child,seen)
    return seen

    def expand(self,value):
        if type(value) is tuple and len(value)==2 and value[0]=='$REF':return self.resolve(value[1])[0]
        if type(value) is dict:return {k:self.expand(v) for k,v in value.items()}

        return copy.deepcopy(value)
    def resolve(self,label):
        if label in self.memo:return copy.deepcopy(self.memo[label])
        assert label in self.defs and label not in self.resolving
        self.resolving.add(label)
        value=self.defs[label]
        raw=value if type(value) is bytes else _wire(self.expand(value))
        result=(_pin(self.paths[label],raw),raw)
        self.memo[label]=result;self.resolving.remove(label)
        return copy.deepcopy(result)
    def finish(self):

        for label in list(self.defs):self.resolve(label)
        records=[dict(pin=copy.deepcopy(pin),body=raw) for pin,raw in self.memo.values()]

def _literal_equal(value,baseline):
    """Strict complete global snapshot comparison; immutable scalar identity fast path."""
    if type(value) is not type(baseline):return False
    if value is baseline:return True
    if type(value) is dict:
        return len(value)==len(baseline) and all(type(k) is str and k in baseline and _literal_equal(v,baseline[k]) for k,v in value.items())
    if type(value) in (list,tuple):
        return len(value)==len(baseline) and all(_literal_equal(a,b) for a,b in zip(value,baseline))
    return value==baseline
